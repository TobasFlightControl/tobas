# Disable the serial console
# Avoid dependency on a specific PARTUUID
# Disable FIQ interrupts
# Avoid RCU (Read-Copy-Update) to be executed on CPU1-3
# Avoid periodical timer interruption to be executed on CPU1-3
# Isolate CPU1-3
# Must leave at least two cores for general-purpose use, or there is a risk of deadlock?
console=tty1 root=/dev/mmcblk0p2 rootfstype=ext4 fsck.repair=yes rootwait cfg80211.ieee80211_regdom=JP dwc_otg.fiq_fsm_enable=0 dwc_otg.fiq_enable=0 dwc_otg.nak_holdoff=0 dwg_otg.speed=1 rcu_nocbs=1-3 nohz_full=1-3 isolcpus=1-3
