# [config.txt | Raspberry Pi Documentation](https://www.raspberrypi.com/documentation/computers/config_txt.html)

[all]

# The kernel will be started in 64-bit mode
arm_64bit=1

# Assign the fast UART to GPIOs 4-5 (/dev/serial0)
dtoverlay=disable-bt

# Allow turbo when low-voltage is present
avoid_warnings=2

# Settings for the performance of processors
# over_voltage=0
# arm_freq=2400
# arm_freq_min=2400
# gpu_freq=910
# force_turbo=0
# ---
over_voltage=6
arm_freq=2800
arm_freq_min=2800
gpu_freq=1000
force_turbo=1

[pi5]

# Allow full current to USB devices
usb_max_current_enable=1

# Enable SPI interfaces
dtoverlay=spi0-2cs
dtoverlay=spi1-3cs

# Enable I2C interface
dtoverlay=i2c1-pi5,baudrate=1000000

# Enable UART interfaces
dtparam=uart0=on
dtoverlay=uart0-pi5
dtoverlay=uart2-pi5
dtoverlay=uart4-pi5
