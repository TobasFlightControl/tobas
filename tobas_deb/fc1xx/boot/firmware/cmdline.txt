# console=tty1: Disable the serial console
# root=PARTUUID=xxxxxxxx-02: Specify the root filesystem partition (must match the contents of /etc/fstab)
# resize: Resize (Expand) the rootfs partition (not filesystem)
# rcu_nocbs=1-3: Avoid RCU (Read-Copy-Update) to be executed on CPU1-3
# nohz_full=1-3: Avoid periodical timer interruption to be executed on CPU1-3
# isolcpus=domain,managed_irq,1-3: Isolate CPU1-3
# irqaffinity=0: Set the default IRQ affinity to CPU0
console=tty1 root=PARTUUID=xxxxxxxx-02 rootfstype=ext4 fsck.repair=yes rootwait cfg80211.ieee80211_regdom=JP resize rcu_nocbs=1-3 nohz_full=1-3 isolcpus=domain,managed_irq,1-3 irqaffinity=0
